OpenAPI Specification

Tip
  • We use Utoipa to generate OpenAPI 3.1 specification from code, by using its macros.
  • We use its jiff_0_2 and uuid features to automatically add support for jiff::{Timestamp, civil::Date} and uuid::Uuid types, the axum_extras feature to enhance Axum integration, and the yaml feature to generate a YAML version of OpenAPI specification.

Add utoipa

Run following cargo add command from the book_service folder.

 cargo add utoipa -F axum_extras,jiff_0_2,uuid,yaml

Add schema Macros

Model

Update book_service/src/models/book.rs.

use jiff::{Timestamp, civil::Date};
use serde::{Deserialize, Serialize};
use utoipa::ToSchema;
use uuid::Uuid;

#[derive(Debug, toasty::Model, Serialize, ToSchema)]
pub struct Book {
    #[auto]
    #[schema(examples("2027-01-01T00:00:00.123456Z"))]
    pub created_at: Timestamp,

    #[auto]
    #[schema(examples("2027-01-01T00:00:00.123456Z"))]
    pub updated_at: Timestamp,

    #[key]
    #[auto]
    #[schema(examples("01bbbbbb-bbbb-7bbb-8bbb-bbbbbbbbbbbb"))]
    pub id: Uuid,

    #[schema(examples("2007-07-21"))]
    pub published_date: Date,

    pub status: BookStatus,

    #[schema(examples("Harry Potter and the Deathly Hallows"))]
    pub title: String,

    #[schema(examples("It is the seventh and final novel in the Harry Potter series"))]
    pub description: Option<String>,

    #[schema(examples(
        "https://upload.wikimedia.org/wikipedia/en/a/a9/Harry_Potter_and_the_Deathly_Hallows.jpg"
    ))]
    pub image_url: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, toasty::Embed, Serialize, Deserialize, ToSchema)]
#[serde(rename_all = "lowercase")]
#[column(type = u8)]
pub enum BookStatus {
    #[column(variant = 0)]
    Pending,
    #[column(variant = 1)]
    Verified,
}

Payload

Update book_service/src/app/book/payload.rs.

use garde::Validate;
use jiff::civil::Date;
use serde::Deserialize;
use utoipa::ToSchema;

use crate::models::BookStatus;

#[derive(Debug, Deserialize, Validate, ToSchema)]
pub struct BookRequest {
    #[garde(length(min = 1, max = 255))]
    #[schema(min_length = 1, max_length = 255, examples("Harry Potter and the Deathly Hallows"))]
    pub title: String,

    #[garde(skip)]
    #[schema(examples("It is the seventh and final novel in the Harry Potter series"))]
    pub description: Option<String>,

    #[garde(url)]
    #[schema(examples(
        "https://upload.wikimedia.org/wikipedia/en/a/a9/Harry_Potter_and_the_Deathly_Hallows.jpg"
    ))]
    pub image_url: Option<String>,

    #[garde(skip)]
    #[schema(examples("2007-07-21"))]
    pub published_date: Date,

    #[garde(skip)]
    pub status: BookStatus,
}

Build API Specification

Merge Module-based OpenAPI Objects

Add a new file book_service/src/openapi.rs.

use utoipa::{OpenApi, openapi::OpenApi as OpenApiDoc};

#[derive(OpenApi)]
#[openapi(
    info(title = "Book Service API", version = "1.0.0"),
    servers(
        (url = "http://localhost:3000", description = "Development")
    ),
    components(schemas(
        crate::models::Book,
        crate::errors::ErrorResponse,
        crate::app::shared::ValidationErrorResponse
    )),
)]
pub struct ApiDoc;

pub fn build_api_doc() -> OpenApiDoc {
    let mut openapi = ApiDoc::openapi();
    openapi.merge(crate::app::book::BookApi::openapi());
    openapi
}

Update book_service/src/lib.rs.

mod openapi;
pub use openapi::build_api_doc;

apidoc Binary

Add a new file book_service/src/bin/apidoc.rs.

use std::{fs::File, io::Write, path::PathBuf};

use book_service::build_api_doc;

fn main() {
    let yaml = build_api_doc().to_yaml().expect("Failed to generate OpenAPI YAML");

    let mut crate_path = PathBuf::from(env!("CARGO_MANIFEST_DIR"));
    crate_path.push("openapi.yaml");

    let mut file = File::create(&crate_path).expect("Failed to create the yaml file");

    file.write_all(yaml.as_bytes()).expect("Failed to write data stream into the yaml file");
}

Update book_service/Cargo.toml to add the apidoc [[bin]].
